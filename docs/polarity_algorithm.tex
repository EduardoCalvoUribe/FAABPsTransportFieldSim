\begin{algorithm}[t]
\caption{Information relay: per-robot rule for building a score gradient toward the target and modulating curvity}
\label{alg:comm}
\KwIn{own position $\mathbf{x}$ and heading $\hat{\mathbf{e}}$; target $\mathbf{g}$; observation range $r$;
      neighbours' positions $\mathbf{x}_j$ and previous scores $s_j^{(t-1)}$}
\KwOut{polarity $\mathbf{p}$, score $s^{(t)}$ and curvity $\kappa$ of this robot}
\BlankLine
$\mathcal{V} \leftarrow$ robots within range $r$ with no wall in between\;
\BlankLine
\uIf{target in line of sight}{
  $\mathbf{p} \leftarrow \unit(\mathbf{g} - \mathbf{x})$\;
  $s^{(t)} \leftarrow 0$\;
}
\uElseIf{$\mathcal{V} = \emptyset$}{
  $\mathbf{p} \leftarrow \mathbf{0}$\;
  $s^{(t)} \leftarrow \infty$\;
}
\Else{
  $s^{*} \leftarrow \min_{j \in \mathcal{V}} s_j^{(t-1)}$\;
  $\mathcal{B} \leftarrow \{\, j \in \mathcal{V} : s_j^{(t-1)} = s^{*} \,\}$\;
  $\mathbf{p} \leftarrow \unit\bigl(\operatorname{mean}_{j \in \mathcal{B}} \mathbf{x}_j - \mathbf{x}\bigr)$\;
  $s^{(t)} \leftarrow s^{*} + 1$\;
}
\BlankLine
$\kappa \leftarrow -\,\hat{\mathbf{e}} \cdot \mathbf{p}$\;
\end{algorithm}
